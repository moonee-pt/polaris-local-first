+/** （）wrapped action lines, used when a reply never marked its spoken lines. */
+function splitStageFromProse(content: string): SpeechSegment[] {
+  const parts: SpeechSegment[] = [];
+  const pattern = /（[^（）]*）|\([^()]*\)/g;
+  let cursor = 0;
+  for (const match of content.matchAll(pattern)) {
+    const start = match.index ?? 0;
+    if (start > cursor) parts.push({ kind: 'speech', text: content.slice(cursor, start) });
+    parts.push({ kind: 'stage', text: match[0] });
+    cursor = start + match[0].length;
+  }
+  if (cursor < content.length) parts.push({ kind: 'speech', text: content.slice(cursor) });
+  return parts;
+}
